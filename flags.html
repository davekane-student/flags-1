regionNames = new Intl.DisplayNames(["en"], { type: "region" })
ukNames = ({
  "gb-eng": "England",
  "gb-nir": "Northern Ireland",
  "gb-sct": "Scotland",
  "gb-wls": "Wales"
})
labelFor = (code) => ukNames[code] ?? regionNames.of(code.toUpperCase()) ?? code

// Dropdown menu to choose a flag and toggle between similar/different
viewof selected = Inputs.select(codes, {
  label: "Select a country or region:",
  format: (code) => `${labelFor(code)} (${code})`,
  value: "us"
})

viewof mode = Inputs.radio(["Most similar", "Most different"], {
  label: "Show 3 flags that are:",
  value: "Most similar"
})

matches = mode === "Most similar" ? similar[selected] : different[selected]

// Display the selected flag and the 3 matching flags
html`<div style="margin-top: 1.5rem;">
  <h3>Selected: ${labelFor(selected)}</h3>
  <img
    src="flags/${selected}.png"
    alt="Flag of ${labelFor(selected)}"
    style="max-width: 320px; width: 100%; border: 1px solid #ddd; border-radius: 4px;"
  />

  <h3 style="margin-top: 2rem;">3 ${mode} Flags (by color, shape, and layout)</h3>
  <div style="display: grid; grid-template-columns: repeat(auto-fit, minmax(200px, 1fr)); gap: 1.5rem; margin-top: 1rem;">
    ${matches.map((code, i) => html`
      <div>
        <h4>#${i + 1}: ${labelFor(code)}</h4>
        <img
          src="flags/${code}.png"
          alt="Flag of ${labelFor(code)}"
          style="width: 100%; border: 1px solid #ddd; border-radius: 4px;"
        />
      </div>
    `)}
  </div>
</div>`
